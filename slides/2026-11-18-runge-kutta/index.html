function ode_theta_linear(A, u0; forcing=zero, tfinal=1, h=0.1, theta=.5)
    u = copy(u0)
    t = 0.
    thist = [t]
    uhist = [u0]
    while t < tfinal
        tnext = min(t+h, tfinal)
        h = tnext - t
        rhs = (I + h*(1-theta)*A) * u .+ h*forcing(t+h*theta)
        u = (I - h*theta*A) \ rhs
        t = tnext
        push!(thist, t)
        push!(uhist, u)
    end
    thist, hcat(uhist...)
end

Rz_theta(z, theta) = (1 + (1 - theta)*z) / (1 - theta*z)

function advdiff_matrix(n; kappa=1, wind=1, upwind=0.)
    dx = 2 / n
    rows = [1]
    cols = [1]
    vals = [0.]
    wrap(j) = (j + n - 1) % n + 1
    for i in 1:n
        append!(rows, [i, i, i])
        append!(cols, wrap.(i-1:i+1))
        diffuse = [-1, 2, -1] * kappa / dx^2
        advect_upwind = [-1, 1, 0] * wind / dx
        advect_center = [-1, 0, 1] * wind / 2dx
        stencil = -diffuse - upwind * advect_upwind - (1 - upwind) * advect_center
        append!(vals, stencil)
    end
    sparse(rows, cols, vals)
end

Stability classes and the θ\thetaθ method¶

Definition: AAA-stability¶

A method is AAA-stable if the stability region

{z:∣R(z)∣≤1}\{ z : |R(z)| \le 1 \}{z:∣R(z)∣≤1}
(1)

contains the entire left half plane

ℜ[z]≤0.\Re[z] \le 0 .ℜ[z]≤0.
(2)

This means that the method can take arbitrarily large time steps without becoming unstable (diverging) for any problem that is indeed physically stable.

Definition: LLL-stability¶

A time integrator with stability function R(z)R(z)R(z) is LLL-stable if

lim⁡z→∞R(z)=0.\lim_{z\to\infty} R(z) = 0 .z→∞lim​R(z)=0.
(3)

For the θ\thetaθ method, we have

lim⁡z→∞1+(1−θ)z1−θz=1−θ−θ.\lim_{z\to \infty} \frac{1 + (1-\theta)z}{1 - \theta z} = \frac{1-\theta}{-\theta} .z→∞lim​1−θz1+(1−θ)z​=−θ1−θ​.
(4)

Evidently only θ=1\theta=1θ=1 is LLL-stable.

Runge-Kutta methods¶

The methods we have considered thus far can all be expressed as Runge-Kutta methods, which are expressed in terms of sss “stage” equations (possibly coupled) and a completion formula. For the ODE

u˙=f(t,u)\dot u = f(t, u)u˙=f(t,u)
(5)

the Runge-Kutta method is

Yi=u(t)+h∑jaijf(t+cjh,Yj)u(t+h)=u(t)+h∑jbjf(t+cjh,Yj)\begin{split} Y_i = u(t) + h \sum_j a_{ij} f(t+c_j h, Y_j) \\ u(t+h) = u(t) + h \sum_j b_j f(t+c_j h, Y_j) \end{split}Yi​=u(t)+hj∑​aij​f(t+cj​h,Yj​)u(t+h)=u(t)+hj∑​bj​f(t+cj​h,Yj​)​
(6)

where ccc is a vector of abscissa, AAA is a table of coefficients, and bbb is a vector of completion weights.

These coefficients are typically expressed in a Butcher Table

[cAbT]=[c0a00a01c1a10a11b0b1].\left[ \begin{array}{c|c} c & A \\ \hline & b^T \end{array} \right] = \left[ \begin{array}{c|cc} c_0 & a_{00} & a_{01} \\ c_1 & a_{10} & a_{11} \\ \hline & b_0 & b_1 \end{array} \right] .[c​AbT​​]=⎣⎡​c0​c1​​a00​a10​b0​​a01​a11​b1​​​⎦⎤​.
(7)

We will see that, for consistency, the abscissa ccc are always the row sums of AAA and that ∑ibi=1\sum_i b_i = 1∑i​bi​=1.

If the matrix AAA is strictly lower triangular, then the method is explicit (does not require solving equations).

function ode_rk_explicit(f, u0; tfinal=1, h=0.1, table=rk4)
    u = copy(u0)
    t = 0.
    n, s = length(u), length(table.c)
    fY = zeros(n, s)
    thist = [t]
    uhist = [u0]
    while t < tfinal
        tnext = min(t+h, tfinal)
        h = tnext - t
        for i in 1:s
            ti = t + h * table.c[i]
            Yi = u + h * sum(fY[:,1:i-1] * table.A[i,1:i-1], dims=2)
            fY[:,i] = f(ti, Yi)
        end
        u += h * fY * table.b[1,:]
        t = tnext
        push!(thist, t)
        push!(uhist, u)
    end
    thist, hcat(uhist...)
end

Runge-Kutta order conditions¶

We consider the autonomous differential equation

u˙=f(u).\dot u = f(u) .u˙=f(u).
(19)

Higher derivatives of the exact soultion can be computed using the chain rule, e.g.,

Undefined control sequence: \dddot at position 59: … f'(u) f(u) \\
\̲d̲d̲d̲o̲t̲ ̲u(t) &= f''(u) …

\begin{align*}
\ddot u(t) &= f'(u) \dot u = f'(u) f(u) \\
\dddot u(t) &= f''(u) f(u) f(u) + f'(u) f'(u) f(u) . \\
\end{align*}

Note that if f(u)f(u)f(u) is linear, f′′(u)=0f''(u) = 0f′′(u)=0. Meanwhile, the numerical solution is a function of the time step hhh,

Yi(h)=u(0)+h∑jaijf(Yj)U(h)=u(0)+h∑jbjf(Yj).\begin{split} Y_i(h) &= u(0) + h \sum_j a_{ij} f(Y_j) \\ U(h) &= u(0) + h \sum_j b_j f(Y_j). \end{split}Yi​(h)U(h)​=u(0)+hj∑​aij​f(Yj​)=u(0)+hj∑​bj​f(Yj​).​
(21)

We will take the limit h→0h\to 0h→0 and equate derivatives of the numerical solution. First we differentiate the stage equations,

\begin{split} Y_i(0) &= u(0) \ \dot Y_i(0) &= \sum_j a_{ij} f(Y_j) \ \ddot Y_i(0) &= 2 \sum_j a_{ij} \dot f(Y_j) \ &= 2 \sum_j a_{ij} f’(Y_j) \dot Y_j \ &= 2\sum_{j,k} a_{ij} a_{jk} f’(Y_j) f(Y_k) \ \dddot Y_i(0) &= 3 \sum_j a_{ij} \ddot f (Y_j) \ &= 3 \sum_j a_{ij} \Big( \sum_k f’‘(Y_j) \dot Y_j \dot Y_k + f’(Y_j) \ddot Y_j \Big) \ &= 3 \sum_{j,k,\ell} a_{ij} a_{jk} \Big( a_{j\ell} f’‘(Y_j) f(Y_k) f(Y_\ell) + 2 a_{k\ell} f’(Y_j) f’(Y_k) f(Y_\ell) \Big) \end{split}

where we have used Liebnitz’s formula for the mmmth derivative,

(hϕ(h))(m)∣h=0=mϕ(m−1)(0).(h \phi(h))^{(m)}|_{h=0} = m \phi^{(m-1)}(0) .(hϕ(h))(m)∣h=0​=mϕ(m−1)(0).
(22)

Similar formulas apply for U˙(0)\dot U(0)U˙(0), U¨(0)\ddot U(0)U¨(0), and \dddot U(0), with bjb_jbj​ in place of aija_{ij}aij​.

Order conditions are nonlinear algebraic equations¶

Equating terms u˙(0)=U˙(0)\dot u(0) = \dot U(0)u˙(0)=U˙(0) yields

∑jbj=1,\sum_j b_j = 1,j∑​bj​=1,

  • Usually a “safety factor” less than 1 is included so the predicted error is less than the threshold to reject a time step.

  • We have used an absolute tolerance above. If the values of solution variables vary greatly in time, a relative tolerance eloc(h)/∥u(t)∥e_{\text{loc}}(h) / \lVert u(t) \rVerteloc​(h)/∥u(t)∥ or a combination thereof is desirable.
