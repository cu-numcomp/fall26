function vander(x, k=nothing)
    if isnothing(k)
        k = length(x)
    end
    m = length(x)
    V = ones(m, k)
    for j in 2:k
        V[:, j] = V[:, j-1] .* x
    end
    V
end

function vander_chebyshev(x, n=nothing)
    if isnothing(n)
        n = length(x) # Square by default
    end
    m = length(x)
    T = ones(m, n)
    if n > 1
        T[:, 2] = x
    end
    for k in 3:n
        #T[:, k] = x .* T[:, k-1]
        T[:, k] = 2 * x .* T[:,k-1] - T[:, k-2]
    end
    T
end

function interp_nearest(x, s)
    A = zeros(length(s), length(x))
    for (i, t) in enumerate(s)
        loc = nothing
        dist = Inf
        for (j, u) in enumerate(x)
            if abs(t - u) < dist
                loc = j
                dist = abs(t - u)
            end
        end
        A[i, loc] = 1
    end
    A
end

Observations¶

  • Piecewise constant interpolation is well conditioned on any set of points

  • Piecewise constant interpolation converges very slowly (needs many points to increase accuracy)

  • Chebyshev/polynomial interpolation requires special input points, otherwise it is ill conditioned

  • Chebyshev/polynomial interpolation has “exponential” convergence for smooth enough functions

How could we improve the accuracy?¶

Cubic splines¶

  • Piecewise cubic function

  • Continuous values

  • Continuous derivatives

Splines¶

If we are given an arbitrary distribution of points, interpolation with a single polynomial is not robust. Piecewise constant interpolation is not very accurate and gives a rough function. We could improve the accuracy by using a piecewise linear function, but the accuracy is still limited and the function still isn’t smooth (there is a “corner” where the slope changes at each data point). Splines are a way to guarantee an arbitrary amount of smoothness. The idea is that given sorted input points {xi}i=0n\{x_i\}_{i=0}^n{xi​}i=0n​, we compute an interpolating polynomial si(x)s_i(x)si​(x) on every interval (xi,xi+1)(x_i, x_{i+1})(xi​,xi+1​).

Interpolation¶

Given a function value yiy_iyi​ at each xix_ixi​, we require

si(xi)=yisi(xi+1)=yi+1\begin{split} s_i(x_i) &= y_i \\ s_i(x_{i+1}) &= y_{i+1} \end{split}si​(xi​)si​(xi+1​)​=yi​=yi+1​​
(1)

so that the polynomial interpolates our data. If the polynomial has order greater than 1, we are left with some extra degrees of freedom. To provide a unique solution, we’ll need to add conditions.

Smoothness¶

The conditions above guarantee continuity, but not smoothness. We use our extra degree(s) of freedom to impose smoothness conditions of the form

si′(xi+1)=si+1′(xi+1)si′′(xi+1)=si+1′′(xi+1).\begin{split} s_i'(x_{i+1}) &= s_{i+1}'(x_{i+1}) \\ s_i''(x_{i+1}) &= s_{i+1}''(x_{i+1}) . \end{split}si′​(xi+1​)si′′​(xi+1​)​=si+1′​(xi+1​)=si+1′′​(xi+1​).​
(2)

These conditions, which are applied at the interior nodes (x=1,…,n−1x=1,\dotsc,n-1x=1,…,n−1) couple the splines from adjacent intervals and causes the spline approximation to be globally coupled.

End-point conditions¶

The conditions above are still not enough to guarantee a unique spline. Suppose we use quadratic polynomials for each sis_isi​. Then with nnn intervals, we have nnn degrees of freedom after imposing the interpolation condition. Meanwhile, there are only n−1n-1n−1 internal nodes. If we impose continuity of the first derivative, we have n−(n−1)=1n - (n-1) = 1n−(n−1)=1 undetermined degrees of freedom. We could fix this by imposing a boundary condition, such as that the slope at one end-point (e.g., s0′(x0)s_0'(x_0)s0′​(x0​)) was equal to a known value. This is not symmetric and is often an unnatural condition.

Suppose we use cubic polynomials. Now we have two degrees of freedom per interval after imposing the interpolation condition. If we impose continuity of the first and second derivatives, we have 2n−2(n−1)=22n - 2(n-1) = 22n−2(n−1)=2 remaining degrees of freedom. A common choice here is the “natural spline”, s0′′(x0)=0s_0''(x_0) = 0s0′′​(x0​)=0 and sn′′(xn)=0s_n''(x_n) = 0sn′′​(xn​)=0. Cubic splines are the most popular spline in this family.

Solving spline interpolation problems¶

We need to choose a basis for the polynomials si(x)s_i(x)si​(x). We could choose

si(x)=ai+bix+cix2+dix3s_i(x) = a_i + b_i x + c_i x^2 + d_i x^3si​(x)=ai​+bi​x+ci​x2+di​x3
(3)

but this would be very ill-conditioned when the interval (xi,xi+1)(x_i,x_{i+1})(xi​,xi+1​) is far from zero. A better-conditioned choice is

si(x)=ai+bi(x−xi)+ci(x−xi)2+di(x−xi)3.s_i(x) = a_i + b_i(x - x_i) + c_i(x - x_i)^2 + d_i(x - x_i)^3 .si​(x)=ai​+bi​(x−xi​)+ci​(x−xi​)2+di​(x−xi​)3.
(4)

The interpolation property gives

ai=yiai+bi(xi+1−xi)+ci(xi+1−xi)2+di(xi+1−xi)3=yi+1\begin{split} a_i &= y_i \\ a_i + b_i(x_{i+1} - x_i) + c_i(x_{i+1} - x_i)^2 + d_i(x_{i+1} - x_i)^3 &= y_{i+1} \end{split}ai​ai​+bi​(xi+1​−xi​)+ci​(xi+1​−xi​)2+di​(xi+1​−xi​)3​=yi​=yi+1​​
(5)

and continuity of the first and second derivatives gives

si′(xi+1)=bi+2ci(xi+1−xi)+3di(xi+1−xi)2=bi+1=si+1′(xi+1)2ci+6di(xi+1−xi)=2ci+1.\begin{split} s_i'(x_{i+1}) = b_i + 2c_i(x_{i+1} - x_i) + 3d_i(x_{i+1}-x_i)^2 &= b_{i+1} = s_{i+1}'(x_{i+1}) \\ 2c_i + 6d_i(x_{i+1} - x_i) &= 2 c_{i+1} . \end{split}si′​(xi+1​)=bi​+2ci​(xi+1​−xi​)+3di​(xi+1​−xi​)22ci​+6di​(xi+1​−xi​)​=bi+1​=si+1′​(xi+1​)=2ci+1​.​
(6)

After trivially eliminating the aia_iai​, this is a block bidiagonal system (3×33\times 33×3 blocks). We can reduce this to a scalar tridiagonal system. Define δi=xi+1−xi\delta_i = x_{i+1} - x_iδi​=xi+1​−xi​ and Δi=yi+1−yi\Delta_i = y_{i+1} - y_iΔi​=yi+1​−yi​. Then eliminate did_idi​ using

di=ci+1−ci3δid_i = \frac{c_{i+1} - c_i}{3\delta_i}di​=3δi​ci+1​−ci​​
(7)

and bib_ibi​ using

biδi=Δi−ciδi2−ci+1−ci3δi⏟diδi3bi=Δiδi−δi3(ci+1+2ci).\begin{split} b_i\delta i &= \Delta_i - c_i\delta_i^2 - \underbrace{\frac{c_{i+1} - c_i}{3\delta_i}}_{d_i} \delta_i^3 \\ b_i &= \frac{\Delta_i}{\delta_i} - \frac{\delta_i}{3}(c_{i+1} + 2c_i) . \end{split}bi​δibi​​=Δi​−ci​δi2​−di​3δi​ci+1​−ci​​​​δi3​=δi​Δi​​−3δi​​(ci+1​+2ci​).​
(8)

Substituting into the equation for continuity of the first derivative gives

Δiδi−δi3(ci+1+2ci)+2ciδi+(ci+1−ci)δi=Δi+1δi+1−δi+13(ci+2+2ci+1)\frac{\Delta_i}{\delta i} - \frac{\delta_i}{3}(c_{i+1} + 2c_i) + 2c_i\delta_i + (c_{i+1} - c_i)\delta_i = \frac{\Delta_{i+1}}{\delta_{i+1}} - \frac{\delta_{i+1}}{3}(c_{i+2} + 2c_{i+1})δiΔi​​−3δi​​(ci+1​+2ci​)+2ci​δi​+(ci+1​−ci​)δi​=δi+1​Δi+1​​−3δi+1​​(ci+2​+2ci+1​)
(9)

which reduces to

δici+2(δi+δi+1)ci+1+δi+1ci+2=3(Δi+1δi+1−Δiδi).\delta_i c_i + 2(\delta_i + \delta_{i+1}) c_{i+1} + \delta_{i+1} c_{i+2} = 3\left(\frac{\Delta_{i+1}}{\delta_{i+1}} - \frac{\Delta_i}{\delta_i}\right) .δi​ci​+2(δi​+δi+1​)ci+1​+δi+1​ci+2​=3(δi+1​Δi+1​​−δi​Δi​​).
(10)

To impose boundary conditions, we add a dummy interval on the right end (the actual value of xn+1>xnx_{n+1}>x_nxn+1​>xn​ cancels out) so that the equation above is valid for i=0,…,n−2i = 0,\dotsc,n-2i=0,…,n−2 and the right boundary condition sn−1′′(xn)=sn′′(xn)s_{n-1}''(x_n) = s_n''(x_n)sn−1′′​(xn​)=sn′′​(xn​) becomes cn=0c_n = 0cn​=0. The left boundary condition s0′′(x0)=0s_0''(x_0) = 0s0′′​(x0​)=0 yields c0=0c_0 = 0c0​=0, so we must solve

[1δ02(δ0+δ1)δ1δ12(δ1+δ2)δ2⋱⋱⋱δn−22(δn−2+δn−1)δn−11][c0c1⋮cn−1cn]=[03(Δ1δ1−Δ0δ0)⋮3(Δn−1δn−1−Δn−2δn−2)0].\begin{bmatrix} 1 & & & & & & \\ \delta_0 & 2(\delta_0+\delta_1) & \delta_1 & & & & \\ & \delta_1 & 2(\delta_1+\delta_2) & \delta_2 & & & \\ & & \ddots & \ddots & \ddots & & \\ & & & & \delta_{n-2} & 2(\delta_{n-2}+\delta_{n-1}) & \delta_{n-1} \\ & & & & & & 1 \\ \end{bmatrix} \begin{bmatrix} c_0 \\ c_1 \\ \vdots \\ \\ c_{n-1} \\ c_n \end{bmatrix} = \begin{bmatrix} 0 \\ 3\left(\frac{\Delta_1}{\delta_1} - \frac{\Delta_0}{\delta_0} \right) \\ \vdots \\ 3\left(\frac{\Delta_{n-1}}{\delta_{n-1}} - \frac{\Delta_{n-2}}{\delta_{n-2}}\right) \\ 0 \end{bmatrix} .⎣⎡​1δ0​​2(δ0​+δ1​)δ1​​δ1​2(δ1​+δ2​)⋱​δ2​⋱​⋱δn−2​​2(δn−2​+δn−1​)​δn−1​1​⎦⎤​⎣⎡​c0​c1​⋮cn−1​cn​​⎦⎤​=⎣⎡​03(δ1​Δ1​​−δ0​Δ0​​)⋮3(δn−1​Δn−1​​−δn−2​Δn−2​​)0​⎦⎤​.

After solving this equation for cic_ici​, we will recover bib_ibi​ and did_idi​, and then can evaluate the spline at arbitrary points.

function spline_interp_and_eval(x, s)
    n = length(x) - 1
    function s_interp(y)
        delta = x[2:end] - x[1:end-1] # diff(x)
        Delta = diff(y)
        T = zeros(n+1, n+1)
        T[1,1] = 1
        for i in 2:n
            T[i, i-1:i+1] = [delta[i-1], 2*(delta[i-1] + delta[i]), delta[i]]
        end
        T[end,end] = 1
        rhs = zeros(n+1)
        rhs[2:end-1] = 3*(Delta[2:end] ./ delta[2:end] - Delta[1:end-1] ./ delta[1:end-1])
        c = T \ rhs
        S = zeros(n, 5)
        S[:, 1] = x[1:end-1]
        S[:, 3] = c[1:end-1]
        S[:, 5] = y[1:end-1]
        S[:, 2] = diff(c) ./ (3 * delta)
        S[:, 4] = Delta ./ delta - delta/3 .* (2*c[1:end-1] + c[2:end])
        S
    end
    
    function polyval(p, x)
        f = p[1]
        for c in p[2:end]
            f = f * x + c
        end
        f
    end

    function s_eval(S, s)
        f = zero(s)
        for (i, t) in enumerate(s)
            left = max(1, searchsortedfirst(S[:,1], t) - 1)
            f[i] = polyval(S[left, 2:end], t - S[left, 1])
        end
        f
    end
    
    A = zeros(length(s), length(x))
    aye = diagm(ones(length(x)))
    for i in 1:length(x)
        e = aye[:, i]
        S = s_interp(e)
        A[:, i] = s_eval(S, s)
    end
    A
end

First, a stable basis!¶

Derivatives of Chebyshev polynomials also satisfy a recurrence.

function chebdiff(x, n=nothing)
    T = vander_chebyshev(x, n)
    m, n = size(T)
    dT = zero(T)
    dT[:,2:3] = [one.(x) 4*x]
    for j in 3:n-1
        dT[:,j+1] = j * (2 * T[:,j] + dT[:,j-1] / (j-2))
    end
    ddT = zero(T)
    ddT[:,3] .= 4
    for j in 3:n-1
        ddT[:,j+1] = j * (2 * dT[:,j] + ddT[:,j-1] / (j-2))
    end
    T, dT, ddT
end

Solving a BVP with Chebyshev collocation¶

A boundary value problem (BVP) asks to find a function u(x)u(x)u(x) satisfying an equation like

−uxx(x)=f(x)-u_{xx}(x) = f(x)−uxx​(x)=f(x)
(15)

subject to boundary conditions u(−1)=au(-1) = au(−1)=a and u′(1)=bu'(1) = bu′(1)=b.

We’ll use the “method of manufactured solutions”: choose u(x)=tanh⁡(2x)u(x) = \tanh(2x)u(x)=tanh(2x) and solve with the corresponding f(x)f(x)f(x). In practice, f(x)f(x)f(x) comes from the physics and you need to solve for u(x)u(x)u(x).

function poisson_cheb(n, rhsfunc, leftbc=(0, zero), rightbc=(0, zero))
    x = CosRange(-1, 1, n)
    T, dT, ddT = chebdiff(x)
    L = -ddT
    rhs = rhsfunc.(x)
    for (index, deriv, func) in
            [(1, leftbc...), (n, rightbc...)]
        L[index,:] = (T, dT)[deriv+1][index,:]
        rhs[index] = func(x[index])
    end
    x, L / T, rhs
end
