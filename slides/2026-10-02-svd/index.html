function vander(x, k=nothing)
    if isnothing(k)
        k = length(x)
    end
    m = length(x)
    V = ones(m, k)
    for j in 2:k
        V[:, j] = V[:, j-1] .* x
    end
    V
end

function gram_schmidt_classical(A)
    m, n = size(A)
    Q = zeros(m, n)
    R = zeros(n, n)
    for j in 1:n
        v = A[:,j]
        R[1:j-1,j] = Q[:,1:j-1]' * v
        v -= Q[:,1:j-1] * R[1:j-1,j]
        R[j,j] = norm(v)
        Q[:,j] = v / R[j,j]
    end
    Q, R
end

function qr_householder(A)
    m, n = size(A)
    R = copy(A)
    V = [] # list of reflectors
    for j in 1:n
        v = copy(R[j:end, j])
        v[1] += sign(v[1]) * norm(v) # <---
        v = normalize(v)
        R[j:end,j:end] -= 2 * v * v' * R[j:end,j:end]
        push!(V, v)
    end
    V, R
end

function qr_chol(A)
    R = cholesky(A' * A).U
    Q = A / R
    Q, R
end

function qr_chol2(A)
    Q, R = qr_chol(A)
    Q, R1 = qr_chol(Q)
    Q, R1 * R
end

Condition numbers for matrices¶

We may have informally referred to a matrix as “ill-conditioned” when the columns are nearly linearly dependent, but let’s make this concept for precise. Recall the definition of (relative) condition number:

κ=max⁡δx∣δf∣/∣f∣∣δx∣/∣x∣.\kappa = \max_{\delta x} \frac{|\delta f|/|f|}{|\delta x|/|x|} .κ=δxmax​∣δx∣/∣x∣∣δf∣/∣f∣​.
(1)

We understood this definition for scalar problems, but it also makes sense when the inputs and/or outputs are vectors (or matrices, etc.) and absolute value is replaced by vector (or matrix) norms. Consider matrix-vector multiplication, for which f(x)=Axf(x) = A xf(x)=Ax.

κ(A)=max⁡δx∥A(x+δx)−Ax∥/∥Ax∥∥δx∥/∥x∥=max⁡δx∥Aδx∥∥δx∥ ∥x∥∥Ax∥=∥A∥∥x∥∥Ax∥.\kappa(A) = \max_{\delta x} \frac{\lVert A (x+\delta x) - A x \rVert/\lVert A x \rVert}{\lVert \delta x\rVert/\lVert x \rVert} = \max_{\delta x} \frac{\lVert A \delta x \rVert}{\lVert \delta x \rVert} \, \frac{\lVert x \rVert}{\lVert A x \rVert} = \lVert A \rVert \frac{\lVert x \rVert}{\lVert A x \rVert} .κ(A)=δxmax​∥δx∥/∥x∥∥A(x+δx)−Ax∥/∥Ax∥​=δxmax​∥δx∥∥Aδx∥​∥Ax∥∥x∥​=∥A∥∥Ax∥∥x∥​.
(2)

∥A∥=max⁡x≠0∥Ax∥∥x∥.\lVert A \rVert = \max_{x \ne 0} \frac{\lVert A x \rVert}{\lVert x \rVert} .∥A∥=x=0max​∥x∥∥Ax∥​.

Condition number of the matrix¶

The condition number of matrix-vector multiplication depends on the vector. The condition number of the matrix is the worst case (maximum) of the condition number for any vector, i.e.,

κ(A)=max⁡x≠0∥A∥∥x∥∥Ax∥.\kappa(A) = \max_{x \ne 0} \lVert A \rVert \frac{\lVert x \rVert}{\lVert A x \rVert} .κ(A)=x=0max​∥A∥∥Ax∥∥x∥​.
(7)

If AAA is invertible, then we can rephrase as

κ(A)=max⁡x≠0∥A∥∥A−1(Ax)∥∥Ax∥=max⁡Ax≠0∥A∥∥A−1(Ax)∥∥Ax∥=∥A∥∥A−1∥.\kappa(A) = \max_{x \ne 0} \lVert A \rVert \frac{\lVert A^{-1} (A x) \rVert}{\lVert A x \rVert} = \max_{A x \ne 0} \lVert A \rVert \frac{\lVert A^{-1} (A x) \rVert}{\lVert A x \rVert} = \lVert A \rVert \lVert A^{-1} \rVert .κ(A)=x=0max​∥A∥∥Ax∥∥A−1(Ax)∥​=Ax=0max​∥A∥∥Ax∥∥A−1(Ax)∥​=∥A∥∥A−1∥.
(8)

Evidently multiplying by a matrix is just as ill-conditioned of an operation as solving a linear system using that matrix.

Least squares and the normal equations¶

A least squares problem takes the form: given an m×nm\times nm×n matrix AAA (m≥nm \ge nm≥n), find xxx such that

∥Ax−b∥\lVert Ax - b \rVert∥Ax−b∥
(13)

is minimized. If AAA is square and full rank, then this minimizer will satisfy Ax−b=0A x - b = 0Ax−b=0, but that is not the case in general because bbb is not in the range of AAA. The residual Ax−bA x - bAx−b must be orthogonal to the range of AAA.

  • Is this the same as saying AT(Ax−b)=0A^T (A x - b) = 0AT(Ax−b)=0?

  • If QR=AQR = AQR=A, is it the same as QT(Ax−b)=0Q^T (A x - b) = 0QT(Ax−b)=0?

We showed that QQTQQ^TQQT is an orthogonal projector onto the range of QQQ. If QR=AQR = AQR=A,

QQT(Ax−b)=QQT(QRx−b)=Q(QTQ)Rx−QQTb=QRx−QQTb=Ax−QQTb.QQ^T (A x - b) = QQ^T(Q R x - b) = Q (Q^T Q) R x - QQ^T b = QR x - QQ^T b = A x - QQ^T b .QQT(Ax−b)=QQT(QRx−b)=Q(QTQ)Rx−QQTb=QRx−QQTb=Ax−QQTb.
(14)

So if bbb is in the range of AAA, we can solve Ax=bA x = bAx=b. If not, we need only orthogonally project bbb into the range of AAA.

The Professional’s Way: QR (Householder)¶

Solve Rx=QTbR x = Q^T bRx=QTb.

  • QR factorization costs 2mn2−23n32 m n^2 - \frac 2 3 n^32mn2−32​n3 operations and is done once per matrix AAA.

  • Computing QTbQ^T bQTb costs 4(m−n)n+2n2=4mn−2n24 (m-n)n + 2 n^2 = 4 mn - 2n^24(m−n)n+2n2=4mn−2n2 (using the elementary reflectors, which are stable and lower storage than naive storage of QQQ).

  • Solving with RRR costs n2n^2n2 operations. Total cost per right hand side is thus 4mn−n24 m n - n^24mn−n2.

This method is stable and accurate.

The 737 MAX Way: Cholesky/Normal Equations¶

fast, unregulated, and dangerous¶

The mathematically equivalent form (ATA)x=ATb(A^T A) x = A^T b(ATA)x=ATb are called the normal equations. The solution process involves factoring the symmetric and positive definite n×nn\times nn×n matrix ATAA^T AATA.

  • Computing ATAA^T AATA costs mn2m n^2mn2 flops, exploiting symmetry.

  • Factoring ATA=RTRA^T A = R^T RATA=RTR costs 13n3\frac 1 3 n^331​n3 flops. The total factorization cost is thus mn2+13n3m n^2 + \frac 1 3 n^3mn2+31​n3.

  • Computing ATbA^T bATb costs 2mn2 m n2mn.

  • Solving with RTR^TRT costs n2n^2n2.

  • Solving with RRR costs n2n^2n2. Total cost per right hand side is thus 2mn+2n22 m n + 2 n^22mn+2n2.

The product ATAA^T AATA is ill-conditioned: κ(ATA)=κ(A)2\kappa(A^T A) = \kappa(A)^2κ(ATA)=κ(A)2 and can reduce the accuracy of a least squares solution.

The Prepper’s Way: Singular Value Decomposition¶

UΣVT=AU \Sigma V^T = AUΣVT=A
(15)

where UUU and VVV have orthonormal columns and Σ\SigmaΣ is diagonal with nonnegative entries. The entries of Σ\SigmaΣ are called singular values and this decomposition is the singular value decomposition (SVD). It may remind you of an eigenvalue decomposition XΛX−1=AX \Lambda X^{-1} = AXΛX−1=A, but

  • the SVD exists for all matrices (including non-square and deficient matrices)

  • U,VU,VU,V have orthogonal columns (while XXX can be arbitrarily ill-conditioned). Indeed, if a matrix is symmetric and positive definite (all positive eigenvalues), then U=VU=VU=V and Σ=Λ\Sigma = \LambdaΣ=Λ. Computing an SVD requires a somewhat complicated iterative algorithm, but a crude estimate of the cost is 2mn2+11n32 m n^2 + 11 n^32mn2+11n3. Note that this is similar to the cost of QRQRQR when m≫nm \gg nm≫n, but much more expensive for square matrices. Solving with the SVD involves

  • Compute UTbU^T bUTb at a cost of 2mn2 m n2mn.

  • Solve with the diagonal n×nn\times nn×n matrix Σ\SigmaΣ at a cost of nnn.

  • Apply VVV at a cost of 2n22 n^22n2. The total cost per right hand side is thus 2mn+2n22 m n + 2n^22mn+2n2.

SVD gives the unique minimum norm solution when AAA is rank deficient¶

Givens Rotation (as example of orthogonal matrix)¶

function givens(theta)
    s = sin(theta)
    c = cos(theta)

Reflection¶

We’ve previously seen that reflectors have the form F=I−2vvTF = I - 2 v v^TF=I−2vvT where vvv is a normalized vector. Reflectors satisfy FTF=IF^T F = IFTF=I and F=FTF = F^TF=FT, thus F2=IF^2 = IF2=I.

function reflect(theta)
    v = [cos(theta), sin(theta)]
    I - 2 * v * v'

Singular Value Decomposition¶

The SVD is A=UΣVTA = U \Sigma V^TA=UΣVT where UUU and VVV have orthonormal columns and Σ\SigmaΣ is diagonal with nonnegative entries. It exists for any matrix (non-square, singular, etc.). If we think of orthogonal matrices as reflections/rotations, this says any matrix can be represented as reflect/rotate, diagonally scale, and reflect/rotate again.
