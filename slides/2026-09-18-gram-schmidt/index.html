using LinearAlgebra

using Polynomials
default(linewidth=4, legendfontsize=12)

function vander(x, k=nothing)
    if isnothing(k)
        k = length(x)
    end
    m = length(x)
    V = ones(m, k)
    for j in 2:k
        V[:, j] = V[:, j-1] .* x
    end
    V
end

Orthogonal matrices¶

If xTy=0x^T y = 0xTy=0 then we say xxx and yyy are orthogonal (or “xxx is orthogonal to yyy”). A vector is said to be normalized if ∥x∥=1\lVert x \rVert = 1∥x∥=1. If xxx is orthogonal to yyy and ∥x∥=∥y∥=1\lVert x \rVert = \lVert y \rVert = 1∥x∥=∥y∥=1 then we say xxx and yyy are orthonormal. A square matrix with orthonormal columns is said to be an orthogonal matrix. We typically use QQQ or UUU and VVV for matrices that are known/constructed to be orthogonal. Orthogonal matrices are always full rank -- the columns are linearly independent. The inverse of a orthogonal matrix is its transpose:

QTQ=QQT=I.Q^T Q = Q Q^T = I .QTQ=QQT=I.
(2)

Orthogonal matrices are a powerful building block for robust numerical algorithms.

Gram-Schmidt orthogonalization¶

Suppose we’re given some vectors and want to find an orthogonal basis for their span.

[a1∣a2]=[q1∣q2][r11r120r22]\Bigg[ a_1 \Bigg| a_2 \Bigg] = \Bigg[ q_1 \Bigg| q_2 \Bigg] \begin{bmatrix} r_{11} & r_{12} \\ 0 & r_{22} \end{bmatrix}[a1​∣∣​a2​]=[q1​∣∣​q2​][r11​0​r12​r22​​]
(3)

A naive algorithm¶

function gram_schmidt_naive(A)
    m, n = size(A)
    Q = zeros(m, n)
    R = zeros(n, n)
    for j in 1:n
        v = A[:,j]
        for k in 1:j-1
            r = Q[:,k]' * v
            v -= Q[:,k] * r
            R[k,j] = r
        end
        R[j,j] = norm(v)
        Q[:,j] = v / R[j,j]
    end
    Q, R
end

function gram_schmidt_classical(A)
    m, n = size(A)
    Q = zeros(m, n)
    R = zeros(n, n)
    for j in 1:n
        v = A[:,j]
        R[1:j-1,j] = Q[:,1:j-1]' * v
        v -= Q[:,1:j-1] * R[1:j-1,j]
        R[j,j] = norm(v)
        Q[:,j] = v / norm(v)
    end
    Q, R
end
