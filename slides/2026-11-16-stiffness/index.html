using LinearAlgebra
using Plots
default(linewidth=4, legendfontsize=12)

struct RKTable
    A::Matrix
    b::Vector
    c::Vector
    function RKTable(A, b)
        s = length(b)
        A = reshape(A, s, s)
        c = vec(sum(A, dims=2))
        new(A, b, c)
    end
end

function rk_stability(z, rk)
    s = length(rk.b)
    1 + z * rk.b' * ((I - z*rk.A) \ ones(s))
end

rk4 = RKTable([0 0 0 0; .5 0 0 0; 0 .5 0 0; 0 0 1 0], [1, 2, 2, 1] / 6)
heun = RKTable([0 0; 1 0], [.5, .5])
Rz_theta(z, theta) = (1 + (1 - theta)*z) / (1 - theta*z)

function ode_rk_explicit(f, u0; tfinal=1, h=0.1, table=rk4)
    u = copy(u0)
    t = 0.
    n, s = length(u), length(table.c)
    fY = zeros(n, s)
    thist = [t]
    uhist = [u0]
    while t < tfinal
        tnext = min(t+h, tfinal)
        h = tnext - t
        for i in 1:s
            ti = t + h * table.c[i]
            Yi = u + h * sum(fY[:,1:i-1] * table.A[i,1:i-1], dims=2)
            fY[:,i] = f(ti, Yi)
        end
        u += h * fY * table.b
        t = tnext
        push!(thist, t)
        push!(uhist, u)
    end
    thist, hcat(uhist...)
end

Forward Euler Method¶

The simplest method for solving y′(t)=f(t,y)y'(t) = f(t,y)y′(t)=f(t,y) is to use numerical differentiation to write

y′≈y(h)−y(0)hy' \approx \frac{y(h) - y(0)}{h}y′≈hy(h)−y(0)​
(5)

which yields the solution estimate

y~(h)=y(0)+hf(0,y(0))\tilde y(h) = y(0) + h f(0, y(0))y~​(h)=y(0)+hf(0,y(0))
(6)

where hhh is the step size. Let’s try this on a scalar problem

y′=−k(y−cos⁡t)y' = -k (y - \cos t)y′=−k(y−cost)
(7)

where kkk is a parameter controlling the rate at which the solution u(t)u(t)u(t) is pulled toward the curve cos⁡t\cos tcost.

function ode_euler(f, y0; tfinal=10., h=0.1)
    y = copy(y0)
    t = 0.
    thist = [t]
    yhist = [y0]
    while t < tfinal
        tnext = min(t+h, tfinal)
        h = tnext - t
        y += h * f(t, y)
        t = tnext
        push!(thist, t)
        push!(yhist, y)
    end
    thist, hcat(yhist...)
end

Linear Stability Analysis¶

Why did Euler diverge (even if slowly) while RK4 solved this problem accurately? And why do both methods diverge if the step size is too large? We can understand the convergence of methods by analyzing the test problem

y′=λyy' = \lambda yy′=λy
(9)

for different values of λ\lambdaλ in the complex plane. One step of the Euler method with step size hhh maps

y↦y+hλy=(1+hλ)⏟R(hλ)y.y \mapsto y + h \lambda y = \underbrace{(1 + h \lambda)}_{R(h \lambda)} y .y↦y+hλy=R(hλ)(1+hλ)​​y.
(10)

Implicit methods¶

Recall that forward Euler is the step

y~(h)=y(0)+hf(0,y(0)).\tilde y(h) = y(0) + h f(0, y(0)) .y~​(h)=y(0)+hf(0,y(0)).

This can be evaluated explicitly; all the terms on the right hand side are known so the approximation y~(h)\tilde y(h)y~​(h) is computed merely by evaluating the right hand side. Let’s consider an alternative, backward Euler (or “implicit Euler”),

The θ\thetaθ method¶

Forward and backward Euler are bookends of the family known as θ\thetaθ methods.

u~(h)=u(0)+hf(θh,θu~(h)+(1−θ)u(0))\tilde u(h) = u(0) + h f\Big(\theta h, \theta\tilde u(h) + (1-\theta)u(0) \Big)u~(h)=u(0)+hf(θh,θu~(h)+(1−θ)u(0))
(16)

which, for linear problems, is solved as

(I−hθA)u(h)=(I+h(1−θ)A)u(0).(I - h \theta A) u(h) = \Big(I + h (1-\theta) A \Big) u(0) .(I−hθA)u(h)=(I+h(1−θ)A)u(0).
(17)

θ=0\theta=0θ=0 is explicit Euler, θ=1\theta=1θ=1 is implicit Euler, and θ=1/2\theta=1/2θ=1/2 are the midpoint or trapezoid rules (equivalent for linear problems). The stability function is

R(z)=1+(1−θ)z1−θz.R(z) = \frac{1 + (1-\theta)z}{1 - \theta z}.R(z)=1−θz1+(1−θ)z​.
(18)

θ\thetaθ method for the oscillator¶

function ode_theta_linear(A, u0; forcing=zero, tfinal=1, h=0.1, theta=.5)
    u = copy(u0)
    t = 0.
    thist = [t]
    uhist = [u0]
    while t < tfinal
        tnext = min(t+h, tfinal)
        h = tnext - t
        rhs = (I + h*(1-theta)*A) * u .+ h*forcing(t+h*theta)
        u = (I - h*theta*A) \ rhs
        t = tnext
        push!(thist, t)
        push!(uhist, u)
    end
    thist, hcat(uhist...)
end

Stability classes and the θ\thetaθ method¶

Definition: AAA-stability¶

A method is AAA-stable if the stability region

{z:∣R(z)∣≤1}\{ z : |R(z)| \le 1 \}{z:∣R(z)∣≤1}
(19)

contains the entire left half plane

ℜ[z]≤0.\Re[z] \le 0 .ℜ[z]≤0.
(20)

This means that the method can take arbitrarily large time steps without becoming unstable (diverging) for any problem that is indeed physically stable.

Definition: LLL-stability¶

A time integrator with stability function R(z)R(z)R(z) is LLL-stable if

lim⁡z→∞R(z)=0.\lim_{z\to\infty} R(z) = 0 .z→∞lim​R(z)=0.
(21)

For the θ\thetaθ method, we have

lim⁡z→∞1+(1−θ)z1−θz=1−θθ.\lim_{z\to \infty} \frac{1 + (1-\theta)z}{1 - \theta z} = \frac{1-\theta}{\theta} .z→∞lim​1−θz1+(1−θ)z​=θ1−θ​.
(22)

Evidently only θ=1\theta=1θ=1 is LLL-stable.

using SparseArrays

function heat_matrix(n)
    dx = 2 / n
    rows = [1]
    cols = [1]
    vals = [0.]
    wrap(j) = (j + n - 1) % n + 1
    for i in 1:n
        append!(rows, [i, i, i])
        append!(cols, wrap.([i-1, i, i+1]))
        append!(vals, [1, -2, 1] ./ dx^2)
    end
    sparse(rows, cols, vals)
end

Advection as a linear ODE¶

function advect_matrix(n; upwind=false)
    dx = 2 / n
    rows = [1]
    cols = [1]
    vals = [0.]
    wrap(j) = (j + n - 1) % n + 1
    for i in 1:n
        append!(rows, [i, i])
        if upwind
            append!(cols, wrap.([i-1, i]))
            append!(vals, [1., -1] ./ dx)
        else
            append!(cols, wrap.([i-1, i+1]))
            append!(vals, [1., -1] ./ 2dx)
        end
    end
    sparse(rows, cols, vals)
end
